# Slider

```tsx
import { ComponentConfig } from "@wecre8websites/strapi-page-builder-react";

export interface MyComponentProps {
  myFieldName: number
}

export const MyComponentConfig: Omit<ComponentConfig<MyComponentProps, MyComponentProps>, "type"> = {
  fields: {
    myFieldName: { type: "slider", label: "My Field Name", min: 0, max: 100, step: 5 },
  },
  defaultProps: {
    myFieldName: 10,
  },
  label: "My Component",
  render: (props) => {
    return <p>{props.myFieldName}</p>
  }
}
```

## Field Definition

| Parameter | Description | Required |
| ---       | ---         | ---      |
| `type` | `slider` | true |
| `label` | The label for the field in the editor. | false |
| `min` | The minimum value for the slider field. | false |
| `max` | The maximum value for the slider field. | false |
| `step` | The step value for the slider field. | false |
